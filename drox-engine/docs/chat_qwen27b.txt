### 1. Le composant corrigé : `section-transition.tsx`

```tsx
"use client";

import React, { useState, useEffect, useRef } from "react";

interface SectionTransitionProps {
  children: React.ReactNode; // Contenu de la section suivante (Section 2)
}

const SectionTransition: React.FC<SectionTransitionProps> = ({ children }) => {
  const [scrollProgress, setScrollProgress] = useState(0);
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleScroll = () => {
      if (!containerRef.current) return;

      const rect = containerRef.current.getBoundingClientRect();
      const windowHeight = window.innerHeight;

      // Calcul de la progression :
      // 0 quand le haut du conteneur est au bas de l'écran
      // 1 quand le bas du conteneur atteint le haut de l'écran
      let progress = -rect.top / (rect.height - windowHeight);

      // Clamp entre 0 et 1
      progress = Math.min(Math.max(progress, 0), 1);

      setScrollProgress(progress);
    };

    window.addEventListener("scroll", handleScroll);

    // Initialiser la position au chargement
    handleScroll();

    return () => {
      window.removeEventListener("scroll", handleScroll);
    };
  }, []);

  // Animation SVG basée sur la progression
  const wavePath1 = `M0,${50 - scrollProgress * 30} Q25,${40 - scrollProgress * 20} 50,${50 - scrollProgress * 30} T100,${50 - scrollProgress * 30}`;
  const wavePath2 = `M0,${60 + scrollProgress * 20} Q25,${70 + scrollProgress * 15} 50,${60 + scrollProgress * 20} T100,${60 + scrollProgress * 20}`;

  const opacity = Math.min(scrollProgress * 2, 1); // Fade in des éléments enfants

  return (
    <div ref={containerRef} className="relative w-full h-[200vh]"> {/* Hauteur étendue pour permettre le scroll */}
      {/* Section 1 (Fixée en haut pendant l'animation) */}
      <div className="sticky top-0 h-screen flex items-center justify-center bg-white z-10 overflow-hidden">
        <h1 className="text-4xl font-bold text-gray-800 mb-8 transition-opacity duration-300" style={{ opacity: 1 - scrollProgress }}>
          Section 1 : Introduction
        </h1>

        {/* SVG Animé */}
        <svg
          viewBox="0 0 120 50"
          preserveAspectRatio="none"
          className="absolute bottom-0 w-full h-64 z-20 transition-all duration-75 ease-linear"
        >
          <path d={wavePath1} fill="#3b82f6" opacity={0.6 + scrollProgress * 0.4}></path>
          <path d={wavePath2} fill="#1d4ed8" opacity={0.8 - scrollProgress * 0.5}></path>

          {/* Éléments décoratifs animés */}
          <circle
            cx="20" cy="30" r="5"
            fill="#fbbf24"
            transform={`translate(0, ${-scrollProgress * 10})`}
            opacity={opacity}
          />
          <rect
            x="80" y="20" width="10" height="10"
            fill="#ef4444"
            transform={`rotate(${scrollProgress * 90}, 85, 25)`}
            opacity={opacity}
          />
        </svg>
      </div>

      {/* Section 2 (Apparaît au fur et à mesure) */}
      <div className="relative z-30 bg-gray-100 min-h-screen pt-32 px-8 transition-all duration-500" style={{ opacity: opacity }}>
        {children}
      </div>
    </div>
  );
};

export default SectionTransition;
```
